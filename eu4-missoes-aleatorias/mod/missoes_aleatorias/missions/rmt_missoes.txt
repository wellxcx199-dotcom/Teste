# Arvores de missoes sorteadas pela decisao rmt_sortear_destino.
#
# Cada bloco de primeiro nivel e uma "serie" (uma coluna na tela de missoes).
# A serie so aparece para quem cumpre o potential, aqui uma flag de destino.
# Dentro dela, cada missao aponta a anterior em required_missions, formando a seta.

rmt_serie_comercio = {
	slot = 5
	generic = no
	ai = no
	has_country_shield = yes
	potential = {
		has_country_flag = rmt_destino_comercio
	}

	rmt_comercio_1 = {
		icon = mission_war_chest
		required_missions = { }
		trigger = {
			home_trade_node = {
				trade_share = {
					country = ROOT
					share = 30
				}
			}
		}
		effect = {
			add_dip_power = 50
			add_prestige = 10
		}
	}

	rmt_comercio_2 = {
		icon = mission_dominate_home_trade_node
		required_missions = { rmt_comercio_1 }
		trigger = {
			home_trade_node = {
				is_strongest_trade_power = ROOT
			}
		}
		effect = {
			add_country_modifier = {
				name = "rmt_mercadores_audazes"
				duration = 9125	# 25 anos
			}
		}
	}

	rmt_comercio_3 = {
		icon = mission_high_income
		required_missions = { rmt_comercio_2 }
		trigger = {
			home_trade_node = {
				trade_share = {
					country = ROOT
					share = 75
				}
			}
		}
		effect = {
			add_years_of_income = 1
			add_prestige = 25
		}
	}
}

rmt_serie_unificacao = {
	slot = 5
	generic = no
	ai = no
	has_country_shield = yes
	potential = {
		has_country_flag = rmt_destino_unificacao
	}

	rmt_unificacao_1 = {
		icon = mission_conquer_50_development
		required_missions = { }
		trigger = {
			grown_by_development = 50
		}
		effect = {
			add_mil_power = 50
			# Reivindicacoes permanentes na regiao que voce deve unificar.
			if = {
				limit = { has_country_flag = rmt_alvo_ita }
				italy_region = {
					limit = { NOT = { owned_by = ROOT } }
					add_permanent_claim = ROOT
				}
			}
			if = {
				limit = { has_country_flag = rmt_alvo_spa }
				iberia_region = {
					limit = { NOT = { owned_by = ROOT } }
					add_permanent_claim = ROOT
				}
			}
		}
	}

	rmt_unificacao_2 = {
		icon = mission_unite_home_region
		required_missions = { rmt_unificacao_1 }
		provinces_to_highlight = {
			OR = {
				AND = {
					region = italy_region
					ROOT = { has_country_flag = rmt_alvo_ita }
				}
				AND = {
					region = iberia_region
					ROOT = { has_country_flag = rmt_alvo_spa }
				}
			}
			NOT = { owned_by = ROOT }
		}
		trigger = {
			OR = {
				AND = {
					has_country_flag = rmt_alvo_ita
					num_of_owned_provinces_with = {
						value = 10
						region = italy_region
					}
				}
				AND = {
					has_country_flag = rmt_alvo_spa
					num_of_owned_provinces_with = {
						value = 15
						region = iberia_region
					}
				}
			}
		}
		effect = {
			add_country_modifier = {
				name = "rmt_ambicao_unificadora"
				duration = 9125
			}
		}
	}

	rmt_unificacao_3 = {
		icon = mission_empire
		required_missions = { rmt_unificacao_2 }
		trigger = {
			OR = {
				AND = {
					has_country_flag = rmt_alvo_ita
					tag = ITA
				}
				AND = {
					has_country_flag = rmt_alvo_spa
					tag = SPA
				}
			}
		}
		effect = {
			add_prestige = 50
			add_adm_power = 100
		}
	}
}
