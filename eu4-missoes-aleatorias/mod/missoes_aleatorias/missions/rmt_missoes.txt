# Arvores de missoes escolhidas no evento rmt_eventos.1.
#
# Cada bloco de primeiro nivel e uma "serie" (uma coluna na tela de missoes).
# A serie so aparece para quem cumpre o potential, aqui uma flag de destino.
# Dentro dela, cada missao aponta a anterior em required_missions, formando a seta.
# A ultima missao de cada destino chama rmt_encerrar_capitulo, que da a
# recompensa final e libera um novo sorteio.
#
# ai = yes so tem efeito se o jogador ligar a decisao "Destinos para a IA".

########################################
# Comercio
########################################
rmt_serie_comercio = {
	slot = 5
	generic = no
	ai = yes
	has_country_shield = yes
	potential = {
		has_country_flag = rmt_destino_comercio
	}

	rmt_comercio_1 = {
		icon = mission_war_chest
		required_missions = { }
		trigger = {
			home_trade_node = {
				trade_share = {
					country = ROOT
					share = 30
				}
			}
		}
		effect = {
			add_dip_power = 50
			add_prestige = 10
		}
	}

	rmt_comercio_2 = {
		icon = mission_dominate_home_trade_node
		required_missions = { rmt_comercio_1 }
		trigger = {
			home_trade_node = {
				is_strongest_trade_power = ROOT
			}
		}
		effect = {
			add_country_modifier = {
				name = "rmt_mercadores_audazes"
				duration = 9125	# 25 anos
			}
		}
	}

	rmt_comercio_3 = {
		icon = mission_high_income
		required_missions = { rmt_comercio_2 }
		trigger = {
			home_trade_node = {
				trade_share = {
					country = ROOT
					share = 75
				}
			}
		}
		effect = {
			rmt_encerrar_capitulo = { destino = comercio }
		}
	}
}

# Colonial (tronco na coluna 5, ramo escolhido por evento na coluna 4)
########################################
rmt_serie_colonial = {
	slot = 5
	generic = no
	ai = yes
	has_country_shield = yes
	potential = {
		has_country_flag = rmt_destino_colonial
	}

	rmt_colonial_1 = {
		icon = mission_establish_high_seas_navy
		required_missions = { }
		trigger = {
			num_of_owned_provinces_with = {
				value = 1
				OR = {
					continent = north_america
					continent = south_america
				}
			}
		}
		effect = {
			add_dip_power = 50
		}
	}

	rmt_colonial_2 = {
		icon = mission_trade_company_region_abroad
		required_missions = { rmt_colonial_1 }
		trigger = {
			num_of_owned_provinces_with = {
				value = 5
				OR = {
					continent = north_america
					continent = south_america
				}
			}
		}
		effect = {
			add_country_modifier = {
				name = "rmt_febre_colonial"
				duration = 9125
			}
			# Protecao: o evento so dispara se o ramo ainda nao foi escolhido.
			if = {
				limit = {
					NOT = { has_country_flag = rmt_ramo_ouro }
					NOT = { has_country_flag = rmt_ramo_povoamento }
				}
				country_event = { id = rmt_eventos.2 }
			}
		}
	}
}

########################################
# Unificacao (Italia ou Espanha)
########################################
rmt_serie_unificacao = {
	slot = 5
	generic = no
	ai = yes
	has_country_shield = yes
	potential = {
		has_country_flag = rmt_destino_unificacao
	}

	rmt_unificacao_1 = {
		icon = mission_conquer_50_development
		required_missions = { }
		trigger = {
			grown_by_development = 50
		}
		effect = {
			add_mil_power = 50
			# Reivindicacoes permanentes na regiao que voce deve unificar.
			if = {
				limit = { has_country_flag = rmt_alvo_ita }
				italy_region = {
					limit = { NOT = { owned_by = ROOT } }
					add_permanent_claim = ROOT
				}
			}
			if = {
				limit = { has_country_flag = rmt_alvo_spa }
				iberia_region = {
					limit = { NOT = { owned_by = ROOT } }
					add_permanent_claim = ROOT
				}
			}
		}
	}

	rmt_unificacao_2 = {
		icon = mission_unite_home_region
		required_missions = { rmt_unificacao_1 }
		provinces_to_highlight = {
			OR = {
				AND = {
					region = italy_region
					ROOT = { has_country_flag = rmt_alvo_ita }
				}
				AND = {
					region = iberia_region
					ROOT = { has_country_flag = rmt_alvo_spa }
				}
			}
			NOT = { owned_by = ROOT }
		}
		trigger = {
			OR = {
				AND = {
					has_country_flag = rmt_alvo_ita
					num_of_owned_provinces_with = {
						value = 10
						region = italy_region
					}
				}
				AND = {
					has_country_flag = rmt_alvo_spa
					num_of_owned_provinces_with = {
						value = 15
						region = iberia_region
					}
				}
			}
		}
		effect = {
			add_country_modifier = {
				name = "rmt_ambicao_unificadora"
				duration = 9125
			}
		}
	}

	rmt_unificacao_3 = {
		icon = mission_empire
		required_missions = { rmt_unificacao_2 }
		trigger = {
			OR = {
				AND = {
					has_country_flag = rmt_alvo_ita
					tag = ITA
				}
				AND = {
					has_country_flag = rmt_alvo_spa
					tag = SPA
				}
			}
		}
		effect = {
			rmt_encerrar_capitulo = { destino = unificacao }
		}
	}
}

# Conquista (alvo escolhido no momento do sorteio, marcado nas provincias)
########################################
rmt_serie_conquista = {
	slot = 5
	generic = no
	ai = no
	has_country_shield = yes
	potential = {
		has_country_flag = rmt_destino_conquista
	}

	rmt_conquista_1 = {
		icon = mission_cannons_firing
		required_missions = { }
		provinces_to_highlight = {
			has_province_flag = rmt_alvo_conquista
			NOT = { owned_by = ROOT }
		}
		trigger = {
			num_of_owned_provinces_with = {
				value = 1
				has_province_flag = rmt_alvo_conquista
			}
		}
		effect = {
			add_mil_power = 50
		}
	}

	rmt_conquista_2 = {
		icon = mission_assemble_an_army
		required_missions = { rmt_conquista_1 }
		provinces_to_highlight = {
			has_province_flag = rmt_alvo_conquista
			NOT = { owned_by = ROOT }
		}
		trigger = {
			num_of_owned_provinces_with = {
				value = 3
				has_province_flag = rmt_alvo_conquista
			}
		}
		effect = {
			add_country_modifier = {
				name = "rmt_furia_conquistadora"
				duration = 3650	# 10 anos
			}
		}
	}

	rmt_conquista_3 = {
		icon = mission_empire
		required_missions = { rmt_conquista_2 }
		provinces_to_highlight = {
			has_province_flag = rmt_alvo_conquista
			NOT = { owned_by = ROOT }
		}
		trigger = {
			custom_trigger_tooltip = {
				tooltip = rmt_conquista_3_tt
				NOT = {
					any_country = {
						has_country_flag = rmt_alvo_pais
						NOT = { is_subject_of = ROOT }
					}
				}
			}
		}
		effect = {
			hidden_effect = { rmt_limpar_alvo_conquista = yes }
			rmt_encerrar_capitulo = { destino = conquista }
		}
	}
}
