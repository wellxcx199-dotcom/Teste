# Decisoes do mod.
#
# Por que uma decisao e nao um evento automatico no inicio do jogo?
# O gatilho de inicio (on_startup) fica num arquivo do jogo original que teria
# de ser copiado e sobrescrito inteiro, o que quebra com atualizacoes e com
# outros mods. Decisoes, ao contrario, podem ficar num arquivo novo so nosso.

country_decisions = {

	# Abre um novo capitulo: sorteia ate 3 cartas e mostra o evento de escolha.
	# Volta a aparecer quando o capitulo atual termina.
	rmt_sortear_destino = {
		major = yes
		potential = {
			NOT = { has_country_flag = rmt_capitulo_ativo }
			NOT = { has_country_flag = rmt_destinos_esgotados }
			OR = {
				ai = no
				AND = {
					has_global_flag = rmt_ia_ativa
					total_development = 150	# so IAs medias e grandes, para poupar desempenho
				}
			}
		}
		allow = {
			is_at_war = no
		}
		effect = {
			custom_tooltip = rmt_sortear_destino_tt
			hidden_effect = {
				set_country_flag = rmt_capitulo_ativo	# tambem serve de cronometro do capitulo
				rmt_sortear_uma_oferta = yes
				rmt_sortear_uma_oferta = yes
				rmt_sortear_uma_oferta = yes
			}
			country_event = { id = rmt_eventos.1 }
		}
		ai_will_do = {
			factor = 1
		}
	}

	# Interruptor da IA. Desligado por padrao.
	rmt_ativar_ia = {
		potential = {
			ai = no
			NOT = { has_global_flag = rmt_ia_ativa }
		}
		allow = {
			always = yes
		}
		effect = {
			set_global_flag = rmt_ia_ativa
		}
		ai_will_do = {
			factor = 0
		}
	}

	rmt_desativar_ia = {
		potential = {
			ai = no
			has_global_flag = rmt_ia_ativa
		}
		allow = {
			always = yes
		}
		effect = {
			clr_global_flag = rmt_ia_ativa
		}
		ai_will_do = {
			factor = 0
		}
	}
}
