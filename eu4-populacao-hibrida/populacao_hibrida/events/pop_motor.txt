# ==========================================================================
#  População Híbrida - evento oculto que executa a simulação
#
#  Disparado por pop_pulso_mensal quando é a vez do país (a cada 2, 3 ou
#  6 meses, conforme o modo). Fica oculto: o jogador nunca o vê, e a IA o
#  resolve automaticamente. Rodar o trabalho num evento, e não dentro do
#  on_action, segue a prática dos mods grandes para evitar travamentos.
# ==========================================================================

namespace = pop_motor

country_event = {
	id = pop_motor.1
	title = pop_motor.1.t
	desc = pop_motor.1.d
	picture = CITY_VIEW_eventPicture
	hidden = yes
	is_triggered_only = yes

	immediate = {
		pop_executar_tick = yes
		# Depuração: os logs ficam AQUI, no evento, e não nos scripted
		# effects. Lá dentro os colchetes de localização travam o jogo.
		if = {
			limit = { has_global_flag = pop_debug }
			log = "POP - [Root.GetName] - provincias processadas neste tick: [Root.pop_provs_tick.GetValue]"
			if = {
				# pop_lote volta a 0 quando o tick fecha um ciclo completo
				limit = { is_variable_equal = { which = pop_lote value = 0 } }
				log = "POP - [Root.GetName] - ciclo nacional: [Root.pop_nac_total.GetValue] mil habitantes, [Root.pop_nac_pct_urbana.GetValue] por cento urbano"
			}
		}
	}

	option = {
		name = pop_motor.1.a
	}
}
