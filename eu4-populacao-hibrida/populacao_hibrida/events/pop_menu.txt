# ==========================================================================
#  População Híbrida - Painel Demográfico (menus por eventos)
#
#  Menus feitos com eventos funcionam sem sobrescrever nenhum arquivo de
#  interface do jogo base, então nunca quebram com patches. Cada opção
#  chama o próximo "menu" com country_event.
#  Todos são is_triggered_only: o jogo nunca os testa sozinho.
# ==========================================================================

namespace = pop_menu

# ---------------------------------------------------------------- Principal
country_event = {
	id = pop_menu.1
	title = pop_menu.1.t
	desc = pop_menu.1.d
	picture = CITY_VIEW_eventPicture
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			pop_somar_totais_nacionais = yes
			pop_calcular_maior_cidade = yes
		}
	}

	option = {	# Camadas de mapa
		name = pop_menu.1.mapas
		country_event = { id = pop_menu.2 }
	}
	option = {	# Desempenho
		name = pop_menu.1.desempenho
		country_event = { id = pop_menu.3 }
	}
	option = {	# Regras
		name = pop_menu.1.regras
		country_event = { id = pop_menu.4 }
	}
	option = {	# Ajuda
		name = pop_menu.1.ajuda
		country_event = { id = pop_menu.5 }
	}
	option = {	# Fechar
		name = pop_menu.1.fechar
	}
}

# --------------------------------------------------------------- Desempenho
country_event = {
	id = pop_menu.3
	title = pop_menu.3.t
	desc = pop_menu.3.d
	picture = ECONOMY_eventPicture
	is_triggered_only = yes

	option = {	# Alto detalhe
		name = pop_menu.3.alto
		trigger = { NOT = { has_global_flag = pop_modo_alto } }
		clr_global_flag = pop_modo_leve
		set_global_flag = pop_modo_alto
		hidden_effect = { pop_recalcular_intervalos = yes }
		country_event = { id = pop_menu.3 }
	}
	option = {	# Equilibrado
		name = pop_menu.3.equilibrado
		trigger = {
			OR = {
				has_global_flag = pop_modo_alto
				has_global_flag = pop_modo_leve
			}
		}
		clr_global_flag = pop_modo_alto
		clr_global_flag = pop_modo_leve
		hidden_effect = { pop_recalcular_intervalos = yes }
		country_event = { id = pop_menu.3 }
	}
	option = {	# Leve
		name = pop_menu.3.leve
		trigger = { NOT = { has_global_flag = pop_modo_leve } }
		clr_global_flag = pop_modo_alto
		set_global_flag = pop_modo_leve
		hidden_effect = { pop_recalcular_intervalos = yes }
		country_event = { id = pop_menu.3 }
	}
	option = {	# IA detalhada: ligar
		name = pop_menu.3.ia_detalhada
		trigger = { NOT = { has_global_flag = pop_ia_detalhada } }
		set_global_flag = pop_ia_detalhada
		hidden_effect = { pop_recalcular_intervalos = yes }
		country_event = { id = pop_menu.3 }
	}
	option = {	# IA simplificada: voltar ao padrão
		name = pop_menu.3.ia_simplificada
		trigger = { has_global_flag = pop_ia_detalhada }
		clr_global_flag = pop_ia_detalhada
		hidden_effect = { pop_recalcular_intervalos = yes }
		country_event = { id = pop_menu.3 }
	}
	option = {
		name = pop_menu.voltar
		country_event = { id = pop_menu.1 }
	}
}
