defined_text = {
	name = GetPopModo
	text = {
		localisation_key = pop_txt_modo_alto
		trigger = { has_global_flag = pop_modo_alto }
	}
	text = {
		localisation_key = pop_txt_modo_leve
		trigger = { has_global_flag = pop_modo_leve }
	}
	text = {
		localisation_key = pop_txt_modo_equilibrado
		trigger = {
			NOT = { has_global_flag = pop_modo_alto }
			NOT = { has_global_flag = pop_modo_leve }
		}
	}
}

# Escopo: país. Depende do alvo salvo em pop_calcular_maior_cidade.
defined_text = {
	name = GetPopMaiorCidade
	text = {
		localisation_key = pop_txt_maior_cidade
		trigger = {
			check_variable = { which = pop_nac_maior_cidade value = 0.001 }
			has_saved_event_target = pop_maior_cidade
		}
	}
	text = {
		localisation_key = pop_txt_sem_dados
		trigger = {
			NOT = {
				AND = {
					check_variable = { which = pop_nac_maior_cidade value = 0.001 }
					has_saved_event_target = pop_maior_cidade
				}
			}
		}
	}
}
